// Helper: Copy Text & Feedback
function copyToClipboard(text, alertMsg = "Copied to clipboard!") {
    navigator.clipboard.writeText(text).then(() => {
        alert(alertMsg);
    }).catch(() => {
        const input = document.createElement("input");
        input.value = text;
        document.body.appendChild(input);
        input.select();
        document.execCommand("copy");
        document.body.removeChild(input);
        alert(alertMsg);
    });
}

function compressImage(file, maxWidth = 800, maxHeight = 800, quality = 0.7) {
    return new Promise((resolve, reject) => {
        if (!file) return resolve("");
        const reader = new FileReader();
        reader.onload = (e) => {
            const img = new Image();
            img.onload = () => {
                const canvas = document.createElement("canvas");
                let width = img.width;
                let height = img.height;

                if (width > height) {
                    if (width > maxWidth) {
                        height = Math.round((height * maxWidth) / width);
                        width = maxWidth;
                    }
                } else {
                    if (height > maxHeight) {
                        width = Math.round((width * maxHeight) / height);
                        height = maxHeight;
                    }
                }

                canvas.width = width;
                canvas.height = height;
                const ctx = canvas.getContext("2d");
                ctx.drawImage(img, 0, 0, width, height);
                resolve(canvas.toDataURL("image/jpeg", quality));
            };
            img.onerror = () => reject("Failed to load image.");
            img.src = e.target.result;
        };
        reader.onerror = () => reject("Failed to read file.");
        reader.readAsDataURL(file);
    });
}

function registerUser(){
    const username = document.getElementById("signupUser").value.trim();
    const email = document.getElementById("signupEmail").value.trim();
    const password = document.getElementById("signupPassword").value;
    const confirm = document.getElementById("signupConfirm").value;
    const refCode = document.getElementById("signupRef").value.trim();
    const message = document.getElementById("signupMessage");

    if(!username || !email || !password){
        message.className = "message error";
        message.textContent = "Please fill all required fields.";
        return;
    }

    if(password !== confirm){
        message.className = "message error";
        message.textContent = "Passwords do not match.";
        return;
    }

    if(users.some(u => u.username.toLowerCase() === username.toLowerCase())){
        message.className = "message error";
        message.textContent = "Username already taken! Choose a different username.";
        return;
    }

    if(users.some(u => u.email.toLowerCase() === email.toLowerCase())){
        message.className = "message error";
        message.textContent = "This Gmail/Email is already registered!";
        return;
    }

    const newUser = {
        id: Date.now(),
        username: username,
        email: email,
        password: password,
        referredBy: refCode,
        balance: 0,
        status: "pending",
        planActive: false,
        planActivatedAt: null,
        lastWeeklyCheck: null,
        weeklyReferralCount: 0,
        lastDailyBonus: null
    };

    users.push(newUser);
    saveData();

    currentUserId = newUser.id;
    isAdmin = false;
    localStorage.removeItem("AE_ADMIN");
    localStorage.setItem("AE_CURRENT_USER", newUser.id);

    document.getElementById("signupUser").value = "";
    document.getElementById("signupEmail").value = "";
    document.getElementById("signupPassword").value = "";
    document.getElementById("signupConfirm").value = "";
    document.getElementById("signupRef").value = "";
    message.textContent = "";

    document.getElementById("signupModal").classList.add("show");
}

function claimDailyBonus(){
    const user = getCurrentUser();
    if(!user) return;

    if(!user.planActive){
        alert("Please activate your lifetime plan first to claim daily bonuses!");
        return;
    }

    const DAY_MS = 24 * 60 * 60 * 1000;
    const now = Date.now();

    if(user.lastDailyBonus && (now - user.lastDailyBonus < DAY_MS)){
        const remainingHours = Math.ceil((DAY_MS - (now - user.lastDailyBonus)) / (1000 * 60 * 60));
        alert(`⏰ Daily bonus already claimed! Next check-in available in ~${remainingHours} hours.`);
        return;
    }

    user.balance += DAILY_BONUS_AMOUNT;
    user.lastDailyBonus = now;
    saveData();

    alert(`🎉 Success! PKR ${DAILY_BONUS_AMOUNT} Daily Bonus added to your balance.`);
    loadDashboard();
}

function checkWeeklyTarget(user){
    if(!user || !user.planActive || !user.lastWeeklyCheck) return;

    const ONE_WEEK_MS = 7 * 24 * 60 * 60 * 1000;
    const now = Date.now();

    if(now - user.lastWeeklyCheck >= ONE_WEEK_MS){
        if(user.weeklyReferralCount < 1){
            const penalty = 250;
            user.balance = Math.max(0, user.balance - penalty);
            alert(`⚠️ Weekly Target Warning: You didn't refer 1 active member this week. PKR ${penalty} deducted from balance.`);
        }
        user.lastWeeklyCheck = now;
        user.weeklyReferralCount = 0;
        saveData();
    }
}

function investInPlan(){
    const user = getCurrentUser();
    if(!user) return;

    if(user.balance < PLAN_PRICE){
        alert(`Insufficient balance! You need PKR ${PLAN_PRICE} in your account balance to activate plan. Please deposit first.`);
        switchTab('depositTab');
        return;
    }

    user.balance -= PLAN_PRICE;
    user.planActive = true;
    user.status = "approved";
    user.planActivatedAt = Date.now();
    user.lastWeeklyCheck = Date.now();
    user.weeklyReferralCount = 0;

    if(user.referredBy){
        const referrer = users.find(u => u.username.toLowerCase() === user.referredBy.toLowerCase());
        if(referrer){
            referrer.balance += REFERRAL_BONUS;
            referrer.weeklyReferralCount = (referrer.weeklyReferralCount || 0) + 1;
        }
    }

    saveData();
    document.getElementById("congratsModal").classList.add("show");
    loadDashboard();
}

function loadMemberTasks(){
    const box = document.getElementById("memberTasksList");
    const user = getCurrentUser();
    if(!user) return;

    const myTasks = tasks.filter(t => t.targetUserId === "all" || t.targetUserId == user.id);

    if(!myTasks.length){
        box.innerHTML = "<p class='small'>No assignments currently available.</p>";
        return;
    }

    box.innerHTML = myTasks.map(t => {
        const sub = submissions.find(s => s.taskId === t.id && s.userId == user.id);
        
        let actionBtn = `<button class="btn primary" style="font-size:12px; padding:6px 12px;" onclick="openTaskModal(${t.id})">Submit Proof</button>`;

        if(sub){
            if(sub.status === "pending") actionBtn = `<span class="status">Under Review</span>`;
            if(sub.status === "approved") actionBtn = `<span class="status approved">Completed ✓</span>`;
            if(sub.status === "rejected") actionBtn = `<span class="status rejected">Rejected ✕</span>`;
        }

        return `
            <div class="task-item">
                <div style="display:flex; justify-content:space-between; align-items:center;">
                    <div>
                        <strong style="font-size:15px;">${safe(t.title)}</strong><br>
                        Reward: <strong style="color:#10b981;">PKR ${t.reward}</strong>
                    </div>
                    <div>${actionBtn}</div>
                </div>
            </div>
        `;
    }).join("");
}

async function submitTaskProof(){
    const taskId = Number(document.getElementById("taskModalId").value);
    const fileInput = document.getElementById("taskProofFile");
    const notes = document.getElementById("taskProofText").value.trim();
    const task = tasks.find(t => t.id === taskId);
    const user = getCurrentUser();

    if(!user) return;

    if(!fileInput.files[0] && !notes){
        alert("Please upload a proof screenshot or provide details in notes.");
        return;
    }

    try {
        let compressedImg = "";
        if(fileInput.files[0]){
            compressedImg = await compressImage(fileInput.files[0]);
        }

        submissions.push({
            id: Date.now(),
            taskId: taskId,
            userId: user.id,
            username: user.username,
            taskTitle: task ? task.title : "Assignment Task",
            reward: task ? task.reward : 0,
            proofImage: compressedImg,
            notes: notes,
            status: "pending",
            createdAt: new Date().toISOString()
        });

        saveData();
        closeTaskModal();
        alert("Proof submitted successfully! Awaiting Admin verification.");
        loadDashboard();
    } catch(err) {
        alert("Error processing image file. Please try another screenshot.");
    }
}

async function submitDeposit(){
    const user = getCurrentUser();
    if(!user) return;

    const amt = Number(document.getElementById("depositAmount").value);
    const tx = document.getElementById("depositTx").value.trim();
    const fileInput = document.getElementById("depositDocument");
    const message = document.getElementById("depositMessage");

    if(!amt || amt < MIN_DEPOSIT){
        message.className = "message error";
        message.textContent = `Minimum deposit amount is PKR ${MIN_DEPOSIT}.`;
        return;
    }

    if(!tx || !fileInput.files[0]){
        message.className = "message error";
        message.textContent = "Please enter TRX ID and upload receipt image.";
        return;
    }

    try {
        message.className = "message";
        message.textContent = "Compressing & uploading receipt...";

        const compressedImg = await compressImage(fileInput.files[0]);

        deposits.push({
            id: Date.now(),
            userId: user.id,
            username: user.username,
            amount: amt,
            transactionId: tx,
            proofImage: compressedImg,
            status: "Pending",
            createdAt: new Date().toISOString()
        });

        saveData();
        message.className = "message success";
        message.textContent = "Deposit request submitted! Reviewing will take up to 24 hours.";
        
        document.getElementById("depositAmount").value = "";
        document.getElementById("depositTx").value = "";
        document.getElementById("depositDocument").value = "";
    } catch(err) {
        message.className = "message error";
        message.textContent = "Error uploading receipt image. Please try another file.";
    }
}

                </div>

                ${d.proofImage ? `<img src="${d.proofImage}" class="screenshot-preview" alt="Proof" onclick="viewFullImage('${d.proofImage}')">` : ''}
                <div class="small" style="color:#6b7280; font-size:11px; margin-top:4px;">${new Date(d.createdAt).toLocaleString()}</div>
            </div>
        `).join("");
    }

function submitWithdrawal(){
    const user = getCurrentUser();
    if(!user) return;

    const amount = Number(document.getElementById("withdrawAmount").value);
    const account = document.getElementById("withdrawAccount").value.trim();
    const message = document.getElementById("withdrawMessage");

    if(!amount || amount < MIN_WITHDRAWAL){
        message.className = "message error";
        message.textContent = `Minimum withdrawal is PKR ${MIN_WITHDRAWAL}.`;
        return;
    }

    if(amount > user.balance){
        message.className = "message error";
        message.textContent = "Insufficient balance.";
        return;
    }

    if(!account){
        message.className = "message error";
        message.textContent = "Please enter account details.";
        return;
    }

    user.balance -= amount;

    withdrawals.push({
        id: Date.now(),
        userId: user.id,
        username: user.username,
        amount: amount,
        account: account,
        status: "Reviewing",
        reason: "",
        createdAt: new Date().toISOString()
    });

    saveData();
    message.className = "message success";
    message.textContent = "Withdrawal request submitted!";
    document.getElementById("withdrawBalance").textContent = "PKR " + user.balance;
    document.getElementById("withdrawAmount").value = "";
    document.getElementById("withdrawAccount").value = "";
}

function declineWithdrawal(requestId){
    const req = withdrawals.find(w => w.id === requestId);
    if(!req) return;

    const reason = prompt("Enter rejection reason:", "Incorrect account details") || "Rejected by admin";
    req.status = "Rejected";
    req.reason = reason;

    const user = users.find(u => u.id === req.userId);
    if(user){ user.balance += req.amount; }

    saveData();
    renderAdmin();
    alert("Withdrawal rejected and PKR " + req.amount + " refunded to user.");
}

function safe(str){
    if(!str) return "";
    return String(str)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}
